/** "5:15–8:15pm", or "11:45am–2:30pm" across noon (L2-222 AC4, the mock's care chip). */
export function compactRange(start: string, end: string): string {
  const samePart = toMinutes(start) >= 12 * 60 === toMinutes(end) >= 12 * 60;
  return `${samePart ? hhmm12(start) : clock12(start)}–${clock12(end)}`;
}

/** "Ada and Ben", "Ada, Ben and Cy". */
export function andList(names: readonly string[]): string {
  if (names.length <= 1) return names.join('');
  return `${names.slice(0, -1).join(', ')} and ${names[names.length - 1]}`;
}

/** "Sunday dinner at Snug Harbour · Ada and Ben" for /past (L2-225 AC7). */
export function pastDateLine(d: DateSummaryDto): string {
  return `${d.day} ${d.meal.toLowerCase()} at ${d.restaurant} · ${andList(d.attendees)}`;
}

WeekendPage.planDate()

protected planDate(): Promise<void> {        // D38, or D38b with fewer than two adults
  return this.run(async () => {
    const weekendId = this.weekend().id;
    const planning = await this.dates.planning(weekendId);
    const ref = this.dialog.open<PlanDateDialogResult, PlanDateDialogData>(PlanDateDialog,
      { ...DIALOG_OPTIONS, data: { planning } });
    const choice = await firstValueFrom(ref.closed);
    if (!choice) return;
    try {
      const date = await this.dates.plan(weekendId, choice.day, choice.meal, choice.attendeeIds);
      await this.router.navigate(['/weekend/dates', date.id]);
    } catch (err) {
      if (!(err instanceof DateRefusal)) throw err;
      this.error.set(err.message);           // the page's warn banner
    }
  });
}

export class PlanDateDialog {
  protected readonly data = inject<PlanDateDialogData>(DIALOG_DATA);
  protected readonly canPlan = this.data.planning.canPlan;
  protected readonly meal = signal<string>('Dinner');
  protected readonly day = signal<string>('Saturday');
  protected readonly going = signal<readonly string[]>(
    this.data.planning.adults.slice(0, 2).map((a) => a.id),
  );
  protected readonly ready = computed(() => this.canPlan && this.going().length === 2);
  protected confirm(): void {
    if (!this.ready()) return;
    this.dialogRef.close({
      day: this.day() as WeekendDay, meal: this.meal() as MealSlot, attendeeIds: this.going(),
    });
  }
}

One change() for every edit

private async change(work: () => Promise<CoupleDateView | null>): Promise<void> {
  this.busy.set(true);          // "Planning your date…"
  this.error.set('');
  try {
    const next = await work();
    if (next) this.date.set(next);
  } catch (err) {
    if (err instanceof DateRefusal) this.error.set(err.message);
    else this.error.set('Something did not go through. Try again in a moment.');
  } finally {
    this.busy.set(false);
  }
}

  readonly slug: PageSlug = "date";

    return this.main.locator(".date-plan .date-step--meal .block__title");
  }
  careBanner(): Locator { return this.main.locator(".date-care"); }
  /** "Swap restaurant" on the meal row; below 720px its chevron. */
  swapRestaurantButton(): Locator {
    const meal = this.main.locator(".date-plan .date-step--meal");
    return meal.getByRole("button", { name: /^Swap restaurant/ })
      .or(meal.locator(".block__chev")).filter({ visible: true });
  }
  cancelButton(): Locator { return control(this.pageActions(), "Cancel date"); }
  backLink(): Locator { return this.pageHeader.getByRole("link", { name: "This weekend" }); }
}

Things to remember

  • The api library carries the DTO, the view, the projections, the contract and the token; pages stay thin
  • "Plan a date" and the date block on Weekend; the date's screen under the same navigation item
  • Every edit is a CDK dialog; a refusal is shown in the server's words
  • Breakfast on Food ideas, the date on Past
  • Page objects own the DOM; six e2e tests and the projection specs prove the behaviour
