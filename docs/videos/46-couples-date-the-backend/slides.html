ADR-025 · the two decisions that matter

1 · A date is its own aggregate

CoupleDate with WeekendId, not flagged itinerary blocks. Attendees, meal, rejected places, regenerate count and its own lifecycle live on it.

2 · The weekend sees one locked block

Kind Date, spanning the date, RefId = the date's id. Every existing path already keeps a locked block fixed.

DateAttendees · DateWindow

public static IReadOnlyList<FamilyMember> Resolve(
    IReadOnlyList<FamilyMember> members, IReadOnlyList<Guid>? requested)
{
    if (requested is null || requested.Count == 0)
    {
        var adults = Adults(members);                       // 18+, oldest first
        if (adults.Count < 2) throw new ValidationException("attendeeMemberIds", NeedsTwoAdults);
        return adults.Take(2).ToList();
    }
    if (requested.Count != 2 || requested[0] == requested[1])
        throw new ValidationException("attendeeMemberIds", "Choose exactly two different people for the date.");
    var chosen = requested.Select(id => members.FirstOrDefault(m => m.Id == id)
        ?? throw new NotFoundException(nameof(FamilyMember), id)).ToList();
    if (chosen.Any(m => m.Age < AdultAge))
        throw new ValidationException("attendeeMemberIds", "Both people on a date must be 18 or over.");
    return chosen.OrderByDescending(m => m.Age).ThenBy(m => m.Name, StringComparer.Ordinal).ToList();
}

/// The meal itself starts and ends inside its window (L2-219).
public static (TimeOnly Start, TimeOnly End) Window(MealSlot meal) => meal switch
{
    MealSlot.Breakfast => (new TimeOnly(8, 0), new TimeOnly(10, 30)),
    MealSlot.Lunch     => (new TimeOnly(11, 30), new TimeOnly(14, 0)),
    _                  => (new TimeOnly(17, 30), new TimeOnly(21, 0)),

Breakfast is a meal slot

// LockRestaurantCommandHandler: a breakfast lock starts at PlannerTimes.BreakfastWindowStart (8:30)
/// Breakfast before 11:00, lunch until 15:00, dinner after (L2-220 AC4).
private static bool IsSlot(TimeOnly startTime, MealSlot slot) => slot switch
{
    MealSlot.Breakfast => startTime < new TimeOnly(11, 0),
    MealSlot.Lunch => startTime >= new TimeOnly(11, 0) && startTime < new TimeOnly(15, 0),
    _ => startTime >= new TimeOnly(15, 0),
};
