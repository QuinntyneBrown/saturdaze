Coding sd-select

A native select in field clothes, wired into Angular forms.

Where it is used

    <sd-select
      label="Roughly how long"
      name="minutes"
      [options]="durations"
      [ngModel]="minutes()"
      (ngModelChange)="minutes.set($event)"
    />

Also add-to-day-dialog: label="When".

Inputs and signals

export class Select implements ControlValueAccessor {
  readonly label = input<string>('');
  readonly options = input<readonly SelectOption[]>([]);
  readonly hint = input<string>('');
  readonly required = input(false, { transform: booleanAttribute });
  readonly name = input<string>('');

  protected readonly id = `sd-select-${nextSelectId++}`;
  protected readonly value = signal<string>('');
  protected readonly disabled = signal<boolean>(false);

  private onChange: (value: string) => void = () => {};
  private onTouched: () => void = () => {};

The methods

  protected handleChange(event: Event): void {
    const v = (event.target as HTMLSelectElement).value;
    this.value.set(v);
    this.onChange(v);
    this.onTouched();
  }

  writeValue(value: string | null): void {
    this.value.set(value ?? '');
  }

The native select

<select
  class="field__input"
  [id]="id"
  [attr.name]="name() || null"
  [disabled]="disabled()"
  [value]="value()"
  (change)="handleChange($event)"
  (blur)="handleChange($event)"
>
  @for (option of options(); track option.value) {
    <option [value]="option.value" [selected]="option.value === value()">{{ option.label }}</option>
  }
</select>
@if (hint()) {
  <p class="field__hint">{{ hint() }}</p>
}

Also: selects the option matching a written value · disables the select from the form

Pitfalls

  • (blur) also calls handleChange — keep form change handlers idempotent
  • No placeholder option: add an explicit empty option if needed
  • Don't swap it for a custom dropdown for looks

Things to remember
