Inputs

  readonly variant = input<ButtonVariant>('primary');
  readonly size = input<ButtonSize>('md');
  readonly type = input<ButtonType>('button');
  /** Full width (`.btn--block`). */
  readonly full = input(false, { transform: booleanAttribute });
  /** Square icon-only button (`.btn--icon`); pair with `label`. */
  readonly icon = input(false, { transform: booleanAttribute });
  …
  /** Toggle state → `aria-pressed`; null leaves the attribute off. */
  readonly pressed = input<boolean | null>(null);
  /** Accessible name; required for icon-only buttons. */
  readonly label = input<string>('');
  /** Hover/focus hint. Defaults to `label` on icon-only buttons; `''` turns it off. */
  readonly tooltip = input<string | null>(null);

classes

  protected readonly classes = computed(() =>
    [
      'btn',
      `btn--${this.variant()}`,
      this.size() !== 'md' ? `btn--${this.size()}` : '',
      this.full() ? 'btn--block' : '',
      this.icon() ? 'btn--icon' : '',
      this.warnText() ? 'btn--warn-text' : '',
      this.btnClass(),
    ]
      .filter(Boolean)
      .join(' '),
  );

Tooltip text, relationship and rel

  protected readonly tooltipText = computed(
    () => this.tooltip() ?? (this.icon() ? this.label() : ''),
  );

  /** Icon-only buttons are named by `aria-label`; the tooltip only describes text ones. */
  protected readonly tooltipRelationship = computed<TooltipRelationship>(() =>
    this.icon() ? 'label' : 'description',
  );

  protected readonly rel = computed(() => (this.target() === '_blank' ? 'noopener' : null));

export function navigateInApp(router: Router, event: MouseEvent, href: string): void {
  if (!isPlainClick(event) || !isInAppHref(href)) return;
  event.preventDefault();
  void router.navigateByUrl(href);
}

The button branch

  <button
    [class]="classes()"
    [attr.type]="type()"
    [disabled]="disabled()"
    [attr.aria-label]="label() || null"
    [attr.aria-pressed]="pressed() === null ? null : pressed()"

The anchor adds href, target, rel, aria-disabled and (click)="onAnchorClick($event)".

Two test helpers

function click(el: HTMLElement, init: MouseEventInit = {}): boolean {
  const event = new MouseEvent('click', { bubbles: true, cancelable: true, ...init });
  let preventedByComponent = false;
  const guard = (e: Event): void => {
    preventedByComponent = e.defaultPrevented;
    e.preventDefault();
  };
  /* … dispatch with a window-level guard … */
  return preventedByComponent;
}

/** Focus as the keyboard would, so `:focus-visible` matches (it keys off the last input event). */
function tabTo(el: HTMLElement): void {
  document.body.dispatchEvent(new KeyboardEvent('keydown', { key: 'Tab', bubbles: true }));
  el.focus();
}

Rendering

  • creates a primary, medium <button type="button"> by default
  • mirrors variant, size, full, icon and warnText to classes
  • disables the inner button
  • exposes pressed as aria-pressed only when it is a toggle
  • uses label as the accessible name

The anchor

  • renders an anchor when href is given
  • routes a plain click on an in-app href through the router
  • lets external, modifier and targeted clicks keep their browser meaning
  • marks a disabled anchor with aria-disabled and swallows its click

Tooltip

  • shows the label on an icon-only button, hidden from assistive tech
  • prefers tooltip text over the label, and an empty tooltip turns it off
  • shows nothing on a text button unless asked, then describes it
fixture.componentRef.setInput('tooltip', 'Adds every block to your calendar');
fixture.detectChanges();
tabTo(inner());
expect(inner().getAttribute('aria-describedby')).toBe(bubble()?.id);

Pitfalls

  • <ng-content> repeated in both branches
  • Icon-only button without a label
  • Defaulting type to submit
  • Styling the host instead of .btn
  • target="_blank" without noopener
