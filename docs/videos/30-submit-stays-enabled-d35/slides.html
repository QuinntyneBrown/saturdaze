  if (!this.attempted()) return [];
  const found: ErrorSummaryProblem[] = [];
  if (this.title().trim().length === 0) {
    found.push({ id: 'title', message: 'Enter a title for the event' });
  }
  ...
  return found;
});

Things to remember

  • A disabled primary action cannot say what is missing
  • Failed press: summary, focus on it, links to fields
  • Derive problems from field values
  • Name the action: Send suggestion, Add member
